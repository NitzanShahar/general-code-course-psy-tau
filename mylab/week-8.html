הצגה גרפית מתקדמת

השבוע נתמקד בהצגה גרפית מתקדמת: גרף טוב הוא לפעמים התוצאה החשובה ביותר במאמר.

ניצור גרפים מסכמים עם טעויות תקן, נציג התפלגויות לפי קבוצות (box plot, violin plot ו-raincloud), נוסיף הערות וחיצים שמדגישים את המסר, נשלב כמה תרשימים בפאנל אחד, ונשמור את התוצאה כקובץ PNG ו-PDF באיכות גבוהה. ב-R נעבוד עם ggplot2 ו-ggdist, ובפייתון עם matplotlib ו-seaborn.

השבוע חל בחנוכה — מפגש הזום של יום ראשון יתקיים כרגיל, אלא אם תימסר הודעה אחרת.

#plot using manual ggplot
df_summary <- df |>
group_by(group, treatment) |>
summarise(
mean = mean(memory_score),
se = sd(memory_score) / sqrt(n()),
)
 
 
 
 
 
ggplot(df, aes(x = group, y = memory_score, color = treatment)) +
 
geom_jitter(position = position_jitter(width = 0.2, height = 0), size = 1, alpha = 0.3) +
 
geom_point(data = df_summary, aes(x = group, y = mean, group = treatment, color = treatment),
position = position_dodge(1), size = 4) +
 
geom_errorbar(data = df_summary,aes(x = group, y = mean, group = treatment, ymin = mean - se, ymax = mean + se, color = treatment ),
position = position_dodge(1), width = 0.2
) +
theme_minimal()

library(patchwork)

df = read.csv("titanic.csv")
df = df |>
  filter(!is.na(Age))
df$Pclass = as.factor(df$Pclass)

# 2. Mean survival by class and sex, with standard errors
df_surv = df |>
  group_by(Pclass, Sex) |>
  summarise(
    mean_surv = mean(Survived),
    se_surv   = sd(Survived) / sqrt(n()),
    .groups   = "drop"
  )

plot1 = ggplot(df_surv, aes(x = Pclass, y = mean_surv, fill = Sex)) +
  geom_col(position = position_dodge(width = 0.9)) +
  geom_errorbar(aes(ymin = mean_surv - se_surv, ymax = mean_surv + se_surv),
                position = position_dodge(width = 0.9), width = 0.2) +
  labs(title = "Survival by class and sex", x = "Class", y = "Survival rate (mean +- SE)")

# 3. Age by class and sex (boxplot)
plot2 = ggplot(df, aes(x = Pclass, y = Age, fill = Sex)) +
  geom_boxplot() +
  labs(title = "Age by class and sex", x = "Class", y = "Age")

# 4. Another plot: age by survival (violin)
plot3 = ggplot(df, aes(x = factor(Survived), y = Age, fill = Sex)) +
  geom_violin() +

plot1 = plot1 +
  annotate("text", x = 1.25, y = 0.7, label = "Most women in 1st\nclass survived") +

#### MULTI-PANEL FIGURE ####

# 2. Layout proportions and a main title
combined_plot = plot1 / (plot2 | plot3) +

combined_plot
