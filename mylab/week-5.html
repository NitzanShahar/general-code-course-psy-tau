import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from sklearn.linear_model import LinearRegression


#### MISSING VALUES ####

# 1. Load the data from last week
df = pd.read_csv("data_updated.csv")

# 2. Add missing values in random rows
df.loc[np.random.choice(df.index, 5, replace=False), "age"] = np.nan
df.loc[np.random.choice(df.index, 2, replace=False), "iq"] = np.nan

# 3. Count missing values
print(df[["age", "iq"]].isna().sum())


#### DESCRIPTIVE STATISTICS ####

# 1. Age statistics (pandas ignores missing values by default)
print("mean:", df["age"].mean())
print("sd:", df["age"].std())
print("median:", df["age"].median())
print("min:", df["age"].min())
print("max:", df["age"].max())


#### LINEAR REGRESSION ####

# 1. Remove rows with NA in age or iq
df_clean = df.dropna(subset=["age", "iq"])

# 2. Unstandardized regression: age predicts iq
model = LinearRegression()
model.fit(df_clean[["age"]], df_clean["iq"])
intercept = model.intercept_
slope = model.coef_[0]
print("intercept:", intercept, "slope:", slope)

# 3. Pearson correlation
print("Pearson r:", df_clean["age"].corr(df_clean["iq"]))


#### PLOT ####

# 1. Scatter plot of age and iq, axes cover the full range of each variable
plt.scatter(df_clean["age"], df_clean["iq"])
plt.xlim(df_clean["age"].min(), df_clean["age"].max())
plt.ylim(df_clean["iq"].min(), df_clean["iq"].max())
plt.title("IQ by age")
plt.xlabel("Age (years)")
plt.ylabel("IQ")

# 2. Regression line added manually from the coefficients
x_line = np.array([df_clean["age"].min(), df_clean["age"].max()])
plt.plot(x_line, intercept + slope * x_line, color="red")
plt.show()
