import os

df = read.csv("data/stroop.csv")

import pandas as pd
df = pd.read_csv("data/stroop.csv")

עליכם השבוע ליצור סקריפט אשר מרחיב את יצירת מבנה הנתונים משיעור שעבר, ומאפשר יצירת סטטיסטיקה תיאורית בסיסית.

חלק א - יצירת הנתונים

  1. יצרו את המשתנים הבאים :
    • משתנה נומרי עם מספר נבדק 1 עד 100 (סדר עולה).
    • את המשתנה גיל דיגמו מהתפלגות אחידה (uniform) בדידה, בין הערכים 15 ל 40.
    • את משתנה המגדר צרו בצורה ראנדומלית (סיכוי שווה).
    • משתנה IQ באמצעות דגימה נורמלית עם ממוצע 100 וס"ת של 15
    • משתנה דיכאון (1 / 0) מהתפלגות ברנולי עם סיכוי של 15% לדיכאון.
    • משתנה חרדה מהתפלגות אחידה (uniform) בין הערכים 0 ל 100.
  2. יצירת מבנה נתונים (זהה לשבוע שעבר):
    • הפיכת המשתנים האלו לאובייקט מבנה נתונים
    • שמירת מבנה הנתונים כקובץ csv

חלק ב - הצגת הנתונים

  1. יצירת סטטיסטיקה תיאורית:
    • הדפיסו את הממוצע, ס"ת, חציון, מינימום ומקסימום למשתנה הגיל בלבד
    • הדפיסו טבלת סטטיסטיקה תיאורית מסכמת עבור כל מבנה הנתונים
  2. הצגה גרפית:
    • צרו תרשים היסטוגרמה לגיל
    • כתבו כותרת age בציר האיקס

שפה: אפשר להגיש ב-R או בפייתון, לבחירתכם. מומלץ לנסות את שתי השפות במהלך הסמסטר.

כותרות הקוד: השתמשו בפורמט הכותרות של הקורס, וכתבו בתחילת הקובץ את שם הקורס, השבוע, מספר תעודת הזהות שלכם ושמכם.

# Foundations of programming for empirical science
# Week 3
# assignment by < First Name > < Last Name >, id < 123456789 >

import numpy as np
import pandas as pd
import matplotlib.pyplot as plt


#### CREATE DATA FRAME ####

# 1. Generate vectors
n_subjects = 100
subject    = np.arange(1, n_subjects + 1)
age        = np.random.randint(15, 41, n_subjects)              # discrete uniform, 15 to 40
gender     = np.random.choice(["male", "female"], n_subjects)
iq         = np.random.normal(100, 15, n_subjects)
depression = np.random.binomial(1, 0.15, n_subjects)            # Bernoulli
anxiety    = np.random.uniform(0, 100, n_subjects)

# 2. Combine vectors into a data frame and save it
df = pd.DataFrame({
    "subject": subject,
    "age": age,
    "gender": gender,
    "iq": iq,
    "depression": depression,
    "anxiety": anxiety
})

df.to_csv("data.csv", index=False)


#### DATA SUMMARY ####

# 1. Descriptive statistics for age
print("mean:", df["age"].mean())
print("sd:", df["age"].std())
print("median:", df["age"].median())
print("min:", df["age"].min())
print("max:", df["age"].max())

# 2. Summary table for the whole data frame
print(df.describe(include="all"))

# 3. Histogram of age
plt.hist(df["age"])
plt.title("Histogram of age")
plt.xlabel("age")
plt.show()
