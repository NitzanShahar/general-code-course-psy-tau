# Foundations of programming for empirical science
# Week 3
# assignment by < First Name > < Last Name >, id < 123456789 >

import numpy as np
import pandas as pd


#### CREATE DATA FRAME ####

# 1. Generate vectors
n_subjects = 100
subject    = np.arange(1, n_subjects + 1)
age        = np.random.choice(np.arange(15, 41), n_subjects)      # discrete uniform, 15 to 40
gender     = np.random.choice(["female", "male"], n_subjects)     # equal chance
iq         = np.random.normal(100, 15, n_subjects)
depression = np.random.binomial(1, 0.15, n_subjects)              # Bernoulli
anxiety    = np.random.uniform(0, 100, n_subjects)

# 2. Combine vectors into a data frame and save it
df = pd.DataFrame({
    "subject": subject,
    "age": age,
    "gender": gender,
    "iq": iq,
    "depression": depression,
    "anxiety": anxiety
})

df.to_csv("data.csv", index=False)
# then download data.csv from the Colab files panel


#### DATA SUMMARY ####

# 1. Descriptive statistics for age
print(df["age"].mean())
print(df["age"].std())
print(df["age"].median())
print(df["age"].min())
print(df["age"].max())

# 2. Summary table for the whole data frame
print(df.describe())

# 3. Histogram of age
ax = df["age"].plot.hist(title="Histogram of age")
ax.set_xlabel("age")
